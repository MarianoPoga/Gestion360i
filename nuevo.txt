function obtenerProveedores() {
  var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
  var sheetCompras = ss.getSheetByName("Compras");
  var sheetPagos = ss.getSheetByName("Pagos Proveedores");
  
  var proveedoresSet = new Set();
  
  // Extraer proveedores de Compras (Columna B / index 1)
  if (sheetCompras && sheetCompras.getLastRow() >= 2) {
    var datosCompras = sheetCompras.getRange(2, 2, sheetCompras.getLastRow() - 1, 1).getValues();
    datosCompras.flat().filter(v => v).forEach(v => proveedoresSet.add(String(v).trim()));
  }
  
  // Extraer proveedores de Pagos Proveedores (Columna B / index 1)
  if (sheetPagos && sheetPagos.getLastRow() >= 2) {
    var datosPagos = sheetPagos.getRange(2, 2, sheetPagos.getLastRow() - 1, 1).getValues();
    datosPagos.flat().filter(v => v).forEach(v => proveedoresSet.add(String(v).trim()));
  }

  var unicos = Array.from(proveedoresSet).sort();
  return unicos;
}

// --- OBTENER SALDOS GLOBALES ---
function obtenerSaldosProveedores() {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hojaCompras = ss.getSheetByName("Compras");
    var hojaPagos = ss.getSheetByName("Pagos Proveedores"); 
    
    var saldos = {}; 
    var totalGlobalDeuda = 0;

    // A. PROCESAR COMPRAS (Deuda / Haber)
    if (hojaCompras && hojaCompras.getLastRow() >= 2) {
      var datosCompras = hojaCompras.getDataRange().getValues();
      // Col B (Proveedor, index 1), Col E (Monto, index 4), Col F (Pago, index 5)
      for (var i = 1; i < datosCompras.length; i++) {
        var proveedor = String(datosCompras[i][1]).trim(); 
        var montoCompra = Number(datosCompras[i][4]) || 0; 
        var tipoPago = String(datosCompras[i][5]).trim(); // << LEEMOS COLUMNA F

        // >> FILTRO CLAVE: Solo si el campo Pago es 'Cuenta Corriente'
        if (proveedor && tipoPago === 'Cuenta Corriente') {
          if (!saldos[proveedor]) saldos[proveedor] = 0;
          saldos[proveedor] += montoCompra; // Aumenta la deuda
        }
      }
    }
    
    // B. PROCESAR PAGOS (Crédito / Debe) - Sin cambios
    if (hojaPagos && hojaPagos.getLastRow() >= 2) {
      var datosPagos = hojaPagos.getDataRange().getValues();
      // Col B (Proveedor, index 1), Col E (Monto de Pago, index 4)
      for (var i = 1; i < datosPagos.length; i++) {
        var proveedor = String(datosPagos[i][1]).trim(); 
        var montoPago = Number(datosPagos[i][4]) || 0; 

        if (proveedor) {
          if (!saldos[proveedor]) saldos[proveedor] = 0;
          saldos[proveedor] -= montoPago; // Disminuye la deuda
        }
      }
    }

    // 1. Convertir mapa a array y sumar la deuda total
    var listaSaldos = Object.keys(saldos).map(function(prov) {
      if (saldos[prov] > 0) { 
        totalGlobalDeuda += saldos[prov];
      }
      return {
        nombre: prov,
        saldo: saldos[prov]
      };
    }).sort(function(a, b) {
        return b.saldo - a.saldo; 
    });

    // 2. Devolver la lista y el total global
    return { listaSaldos: listaSaldos, totalGlobalDeuda: totalGlobalDeuda }; 
  } catch (e) {
    return { error: "Error obteniendo saldos: " + e.message, listaSaldos: [], totalGlobalDeuda: 0 };
  }
}


// --- OBTENER HISTORIAL DETALLADO ---
function obtenerHistorialProveedor(nombreProveedor) {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hojaCompras = ss.getSheetByName("Compras");
    var hojaPagos = ss.getSheetByName("Pagos Proveedores"); 
    var zona = ss.getSpreadsheetTimeZone();
    var nombreBuscado = String(nombreProveedor).toLowerCase().trim();
    
    var todasLasTransacciones = []; 

    // A. PROCESAR COMPRAS
    if (hojaCompras && hojaCompras.getLastRow() >= 2) {
      var datosCompras = hojaCompras.getDataRange().getValues();
      // Col A (Fecha, index 0), Col B (Proveedor, index 1), Col D (Detalle, index 3), Col E (Monto, index 4), Col F (Pago, index 5)
      for (var i = 1; i < datosCompras.length; i++) {
        var proveedor = String(datosCompras[i][1]).toLowerCase().trim();
        var tipoPago = String(datosCompras[i][5]).trim(); // << LEEMOS COLUMNA F

        // >> FILTRO CLAVE: Solo si el proveedor coincide Y el campo Pago es 'Cuenta Corriente'
        if (proveedor === nombreBuscado && tipoPago === 'Cuenta Corriente') {
          Logger.log("¡TRANSACCIÓN ENCONTRADA!: Monto: " + monto + ", Fecha: " + fecha);
          var fecha = datosCompras[i][0];
          var detalle = datosCompras[i][3]; // Columna Detalle
          var monto = Number(datosCompras[i][4]) || 0; 

          if (monto >= 0) {
            todasLasTransacciones.push({
              fechaObj: (fecha instanceof Date) ? fecha : new Date(fecha),
              fecha: (fecha instanceof Date) ? Utilities.formatDate(fecha, zona, "dd/MM/yyyy") : fecha,
              detalle: detalle || "Compra a CC",
              monto: monto, 
              tipo: 'DEUDA', 
              orden: 1 
            });
          }
        }
      }
    }

    // B. PROCESAR PAGOS - Sin cambios
    if (hojaPagos && hojaPagos.getLastRow() >= 2) {
      var datosPagos = hojaPagos.getDataRange().getValues();
      // Col A (Fecha Reg, index 0), Col B (Proveedor, index 1), Col C (Medio, index 2), Col E (Monto Pago, index 4)
      for (var i = 1; i < datosPagos.length; i++) {
        var proveedor = String(datosPagos[i][1]).toLowerCase().trim();
        if (proveedor === nombreBuscado) {
          var fecha = datosPagos[i][0]; 
          var medio = datosPagos[i][2]; // Detalle/Medio de Pago
          var monto = Number(datosPagos[i][4]) || 0; 

          if (monto > 0) {
            todasLasTransacciones.push({
              fechaObj: (fecha instanceof Date) ? fecha : new Date(fecha), 
              fecha: (fecha instanceof Date) ? Utilities.formatDate(fecha, zona, "dd/MM/yyyy") : fecha,
              detalle: "Pago (" + medio + ")",
              monto: -monto, 
              tipo: 'PAGO',
              orden: 2 
            });
          }
        }
      }
    }
    
    // C. ORDENAR y CALCULAR SALDO
    todasLasTransacciones.sort(function(a, b) {
      if (a.fechaObj.getTime() !== b.fechaObj.getTime()) {
        return a.fechaObj.getTime() - b.fechaObj.getTime();
      }
      return a.orden - b.orden;
    });

    var saldoAcumulado = 0;
    var movimientosConSaldo = [];

    // Calcular el saldo acumulado
    todasLasTransacciones.forEach(function(mov) {
      saldoAcumulado += mov.monto; 
      movimientosConSaldo.push({
        fecha: mov.fecha,
        detalle: mov.detalle,
        monto: Math.abs(mov.monto), 
        tipo: mov.tipo,
        saldo: saldoAcumulado 
      });
    });
    
    // Devolvemos el saldo actual y los movimientos (los invertimos para ver lo más nuevo arriba en la UI)
    return {
      movimientos: movimientosConSaldo.reverse(),
      total: saldoAcumulado
    };

  } catch (e) {
    return { error: "Error cargando historial: " + e.message };
  }
}

function registrarPago(formObject) {
  // Esta función no necesita cambios
  var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
  var sheet = ss.getSheetByName("Pagos Proveedores"); 
  
  if (!sheet) throw new Error("La hoja 'Pagos Proveedores' no existe. Por favor, créala.");

  var parts = formObject.fechaRegistro.split('-');
  var fechaReg = new Date(parts[0], parts[1] - 1, parts[2]); 
  
  var fechaPago = "";
  if (formObject.fechaPago) {
    var partsP = formObject.fechaPago.split('-');
    fechaPago = new Date(partsP[0], partsP[1] - 1, partsP[2]);
  }

  var montoPago = parseFloat(formObject.monto);
  var proveedorActual = formObject.proveedor;
  
  var observacion = "Pago ERP"; 
  
  sheet.appendRow([
    fechaReg,                
    proveedorActual,    
    formObject.medio,        
    fechaPago,               
    montoPago,       
    0,               
    observacion      
  ]);
  
  var lastRow = sheet.getLastRow();
  sheet.getRange(lastRow, 5, 1, 1).setNumberFormat('$ #,##0.00'); 
  
  return "Éxito";
}

function obtenerHistorialEmpleado(nombreEmpleado) {
  var respuesta = { movimientos: [], total: 0, saldoAnterior: 0 };
  
  try {
    var ss = SpreadsheetApp.openById(ID_RRHH);
    var hoja = ss.getSheetByName("CtaCte Empleados");
    
    if (hoja && hoja.getLastRow() > 1) {
      var datos = hoja.getDataRange().getValues();
      var zona = ss.getSpreadsheetTimeZone();
      var nombreBuscado = String(nombreEmpleado).toLowerCase().trim();
      
      // 1. Recolectar TODA la historia cronológica
      var historiaCompleta = [];
      
      for (var i = 1; i < datos.length; i++) {
        var nombreEnFila = String(datos[i][1]).toLowerCase().trim();
        
        if (nombreEnFila == nombreBuscado) {
          var fechaRaw = datos[i][0];
          var fechaObj = null;
          
          if (fechaRaw instanceof Date) {
            fechaObj = fechaRaw;
          } else if (typeof fechaRaw === 'string') {
             var p = fechaRaw.split("/"); 
             if(p.length === 3) fechaObj = new Date(p[2], p[1]-1, p[0]);
          }

          if (!fechaObj) continue;

          var concepto = String(datos[i][2]); 
          var debe = Number(datos[i][3]) || 0;  
          var haber = Number(datos[i][4]) || 0; 
          var medio = datos[i][5]; 
          
          historiaCompleta.push({
            fechaFmt: Utilities.formatDate(fechaObj, zona, "dd/MM/yyyy"),
            concepto: concepto,
            debe: debe,
            haber: haber,
            medio: medio
          });
        }
      }
      
      // 2. Buscar el índice de CORTE (El último "Pago Sueldo" o "Pago Mes")
      var indiceCorte = -1; 
      
      for (var j = historiaCompleta.length - 1; j >= 0; j--) {
        var c = historiaCompleta[j].concepto.toLowerCase();
        // Lógica de corte: busca "pago" + ("mes" o "sueldo")
        if (c.includes("pago") && (c.includes("mes") || c.includes("sueldo"))) {
           indiceCorte = j; 
           break; 
        }
      }

      // 3. Calcular Saldo Anterior (Suma de todo HASTA el corte inclusive)
      for (var k = 0; k <= indiceCorte; k++) {
         var mov = historiaCompleta[k];
         respuesta.saldoAnterior += (mov.haber - mov.debe);
      }

      // 4. Armar la lista visible (Solo lo que pasó DESPUÉS del corte)
      for (var k = indiceCorte + 1; k < historiaCompleta.length; k++) {
         var mov = historiaCompleta[k];
         
         var tipoMov = (mov.haber > 0) ? "ACREDITADO" : "PAGADO";
         var montoVisual = (mov.haber > 0) ? mov.haber : mov.debe;
         
         respuesta.movimientos.push({
           fecha: mov.fechaFmt,
           concepto: mov.concepto,
           tipo: tipoMov,
           monto: montoVisual,
           medio: mov.medio
         });
      }
      
      // 5. Total Global Real
      var saldoNuevos = 0;
      respuesta.movimientos.forEach(m => {
         if (m.tipo === "ACREDITADO") saldoNuevos += m.monto;
         else saldoNuevos -= m.monto;
      });

      respuesta.total = respuesta.saldoAnterior + saldoNuevos;
    }
    return respuesta;

  } catch (e) { 
    return { error: "Error Backend: " + e.message, movimientos: [], total: 0 }; 
  }
}

// --- GUARDAR MOVIMIENTO (Sueldo, Pago o Ambos) ---
function registrarMovimientoRRHH(formObject) {
  try {
    var ssRRHH = SpreadsheetApp.openById(ID_RRHH);
    var hojaRRHH = ssRRHH.getSheetByName("CtaCte Empleados");
    
    if (!hojaRRHH) {
      hojaRRHH = ssRRHH.insertSheet("CtaCte Empleados");
      hojaRRHH.appendRow(["Fecha", "Empleado", "Concepto", "Debe (Pagos)", "Haber (Sueldos)", "Medio Pago"]);
      hojaRRHH.getRange("A1:F1").setFontWeight("bold");
    }
    
    var fecha = new Date();
    var monto = parseFloat(formObject.monto);
    var empleado = formObject.empleado;
    var concepto = formObject.concepto; 
    var medioPago = formObject.medioPago;
    var tipoOperacion = formObject.tipoOperacion; 
    
    // --- LÓGICA DE GUARDADO ---
    
    if (tipoOperacion === "SueldoPago") {
      // CASO DOBLE ENTRADA: DOS FILAS SEPARADAS
      
      // 1. Fila de devengado (SUELDO/HABER)
      // Concepto original, va al HABER, sin medio de pago
      hojaRRHH.appendRow([fecha, empleado, concepto, 0, monto, "-"]);
      
      // 2. Fila de pago (PAGO/DEBE)
      // Agregamos "Pago " al concepto, va al DEBE, con el medio de pago real
      hojaRRHH.appendRow([fecha, empleado, "Pago " + concepto, monto, 0, medioPago]);

      // Formatear las últimas 2 filas agregadas
      var lastRow = hojaRRHH.getLastRow();
      var rango = hojaRRHH.getRange(lastRow - 1, 1, 2, 6); 
      rango.offset(0,0,2,1).setNumberFormat("dd/MM/yyyy"); 
      rango.offset(0,3,2,2).setNumberFormat("$ #,##0.00"); 

      // Impactar Caja (Solo por la parte del pago)
      if (medioPago === "Rendición" || medioPago === "Efectivo" || medioPago === "Caja") {
         impactarCaja(empleado, "Pago " + concepto, monto, medioPago);
      }

    } else {
      // CASO NORMAL (UNA SOLA FILA: O DEBE O HABER)
      var valorDebe = 0;
      var valorHaber = 0;

      if (tipoOperacion === "Haber") {
        valorHaber = monto;
      } else {
        valorDebe = monto; 
      }

      hojaRRHH.appendRow([fecha, empleado, concepto, valorDebe, valorHaber, medioPago]);
      
      // Formato última fila
      var lastRow = hojaRRHH.getLastRow();
      hojaRRHH.getRange(lastRow, 1).setNumberFormat("dd/MM/yyyy");
      hojaRRHH.getRange(lastRow, 4, 1, 2).setNumberFormat("$ #,##0.00");

      // Impactar Caja (Solo si es DEBE y sale dinero)
      if (tipoOperacion === "Debe" && (medioPago === "Rendición" || medioPago === "Efectivo" || medioPago === "Caja")) {
         impactarCaja(empleado, concepto, monto, medioPago);
      }
    }

    return "Éxito";

  } catch (e) { return "Error: " + e.message; }
}

// --- PROCESAMIENTO PDF ---
function procesarTextoPDF(texto) {
  var log = [];
  try {
    var resultado = { periodo: "Desconocido", empleadosEncontrados: [] };
    
    // --- CAMBIO AQUÍ: Mejor detección y formato MM-AAAA ---
    var matchPeriodo = texto.match(/(\d{1,2})\s*[-/]\s*(\d{4})\s+Haberes/i);
    if (matchPeriodo) {
       // padStart(2, '0') asegura que sea '05-2025' y no '5-2025'
       var mes = matchPeriodo[1].padStart(2, '0');
       var anio = matchPeriodo[2];
       resultado.periodo = mes + "-" + anio;
    }
    // -----------------------------------------------------
    
    var mapaEmpleados = {}; 
    var listaOpciones = []; 
    
    var ss = SpreadsheetApp.openById(ID_RRHH); 
    var hojaP = ss.getSheetByName("Personal");
    
    if (hojaP) {
       var datosP = hojaP.getDataRange().getValues();
       for (var k = 1; k < datosP.length; k++) {
          var apodo = datosP[k][0]; 
          var cuitRaw = String(datosP[k][4]); 
          var activo = datosP[k][6]; 
          
          if (apodo && activo === true) {
             listaOpciones.push(apodo);
             var cuitLimpio = cuitRaw.replace(/[^0-9]/g, ''); 
             if (cuitLimpio.length > 5) mapaEmpleados[cuitLimpio] = apodo;
          }
       }
    }
    listaOpciones.sort();

    var bloques = texto.split(/Legajo N[º°]?:/i); 
    
    for (var i = 1; i < bloques.length; i++) {
      var bloque = bloques[i];
      if (!bloque.includes("TOTALES DEL LEGAJO")) continue; 

      var matchNombre = bloque.match(/Apellido y nombres:\s*([^\r\n]+?)(?=\s+(?:CUIL|CUIT|Tipo))/i);
      var nombrePDF = matchNombre ? matchNombre[1].trim().replace(/\.{2,}/g, "") : "❌ NO NOMBRE";

      var matchCuil = bloque.match(/(?:CUIL|CUIT):\s*([\d\-\s.]{11,15})/i); 
      var cuilPDF = matchCuil ? matchCuil[1] : "";
      var cuilLimpio = cuilPDF.replace(/[^0-9]/g, '');

      var montoNeto = 0;
      var matchLineaTotales = bloque.match(/TOTALES DEL LEGAJO:(.*?)(\r\n|\r|\n|$)/i); 
      
      if (matchLineaTotales) {
         var lineaNumeros = matchLineaTotales[1].trim();
         var partesLinea = lineaNumeros.split(/\s+/);
         var ultimoValor = partesLinea[partesLinea.length - 1]; 
         if (ultimoValor) {
            var soloNumeros = ultimoValor.replace(/[^0-9]/g, '');
            montoNeto = parseFloat(soloNumeros) / 100;
         }
      }

      var nombreSugerido = "";
      if (cuilLimpio && mapaEmpleados[cuilLimpio]) {
         nombreSugerido = mapaEmpleados[cuilLimpio];
      }
      
      if (montoNeto > 0) {
        resultado.empleadosEncontrados.push({
          nombrePDF: nombrePDF,
          cuilPDF: cuilLimpio, 
          nombreDB: nombreSugerido, 
          monto: montoNeto
        });
      }
    }
    
    return { status: "OK", data: resultado, listaOpciones: listaOpciones, log: log.join("\n") };

  } catch (e) {
    return { status: "Error", message: "Error al procesar: " + e.message };
  }
}

function generarTextoRecibo(nombreEmpleado) {
  try {
    var ss = SpreadsheetApp.openById(ID_RRHH);
    var hoja = ss.getSheetByName("CtaCte Empleados");
    var datos = hoja.getDataRange().getValues();
    var nombreBuscado = String(nombreEmpleado).toLowerCase().trim();
    var zona = ss.getSpreadsheetTimeZone();

    // Buscar Teléfono
    var telefono = "";
    var hojaPersonal = ss.getSheetByName("Personal");
    if (hojaPersonal) {
      var datosP = hojaPersonal.getDataRange().getValues();
      for (var p = 1; p < datosP.length; p++) {
        if (String(datosP[p][0]).toLowerCase().trim() == nombreBuscado) {
           var rawTel = String(datosP[p][8]); 
           telefono = rawTel.replace(/[^0-9]/g, "");
           if (telefono.length === 10) telefono = "549" + telefono;
           break;
        }
      }
    }

    // Obtener historia
    var historia = [];
    for (var i = 1; i < datos.length; i++) {
      if (String(datos[i][1]).toLowerCase().trim() == nombreBuscado) {
        historia.push({
          fecha: datos[i][0],
          concepto: String(datos[i][2]),
          debe: Number(datos[i][3]) || 0,
          haber: Number(datos[i][4]) || 0
        });
      }
    }

    if (historia.length === 0) return { texto: "No hay datos.", telefono: "" };

    // Identificar ciclo
    var itemsCiclo = [];
    var ultimoSueldoIdx = -1;
    
    for (var j = historia.length - 1; j >= 0; j--) {
      var c = historia[j].concepto.toLowerCase();
      if (c.includes("sueldo") || c.includes("haberes")) {
        ultimoSueldoIdx = j;
        break;
      }
    }
    if (ultimoSueldoIdx === -1) ultimoSueldoIdx = historia.length - 1;

    for (var j = ultimoSueldoIdx; j >= 0; j--) {
      var c = historia[j].concepto.toLowerCase();
      if (j !== ultimoSueldoIdx && (c.includes("sueldo") || c.includes("haberes"))) {
        break;
      }
      itemsCiclo.unshift(historia[j]); 
    }

    // Calcular Saldos
    var saldoFinalReal = 0;
    historia.forEach(m => saldoFinalReal += (m.haber - m.debe));

    var sumaIngresos = 0;
    var sumaEgresos = 0;
    itemsCiclo.forEach(m => {
       sumaIngresos += m.haber;
       sumaEgresos += m.debe;
    });

    var saldoAnterior = saldoFinalReal - sumaIngresos + sumaEgresos;

    // --- ARMADO DEL TEXTO CON FORMATO ---
    var txt = "🧾 *RECIBO DE HABERES*\n";
    txt += "👤 " + nombreEmpleado.toUpperCase() + "\n";
    txt += "📅 " + Utilities.formatDate(new Date(), zona, "dd/MM/yyyy") + "\n";
    txt += "--------------------------------\n";
    
    txt += "*INGRESOS*\n";
    var hayIngresos = false;
    itemsCiclo.forEach(m => {
      if (m.haber > 0) {
        var fecha = Utilities.formatDate(new Date(m.fecha), zona, "dd/MM");
        // USO DE formatearNumero AQUÍ
        txt += `${fecha} ${m.concepto.padEnd(18).substring(0,18)} ${formatearNumero(m.haber)}\n`;
        hayIngresos = true;
      }
    });
    if(!hayIngresos) txt += "(Sin ingresos)\n";
    txt += "--------------------------------\n";
    txt += `*TOTAL INGRESOS: ${formatearNumero(sumaIngresos)}*\n\n`;

    txt += "*DESCUENTOS / ADELANTOS*\n";
    var hayEgresos = false;
    itemsCiclo.forEach(m => {
      if (m.debe > 0) {
        var fecha = Utilities.formatDate(new Date(m.fecha), zona, "dd/MM");
        // Agregamos el signo menos manual antes del formateo
        txt += `${fecha} ${m.concepto.padEnd(18).substring(0,18)} -${formatearNumero(m.debe).replace("$ ","")}\n`;
        hayEgresos = true;
      }
    });
    if(!hayEgresos) txt += "(Sin adelantos)\n";
    txt += "--------------------------------\n";
    // Total Egresos en negativo
    txt += `*TOTAL EGRESOS: -${formatearNumero(sumaEgresos).replace("$ ","")}*\n\n`;

    txt += "================================\n";
    if (Math.abs(saldoAnterior) > 1) {
       txt += `Saldo Anterior:     ${formatearNumero(saldoAnterior)}\n`;
    }
    txt += `*NETO A PAGAR:       ${formatearNumero(saldoFinalReal)}*\n`;
    txt += "================================";

    return { texto: txt, telefono: telefono };

  } catch (e) {
    return { texto: "Error: " + e.message, telefono: "" };
  }
}

// --- FUNCIÓN DE FORMATO DE MONEDA ---
function formatearNumero(n) {
  if (!n && n !== 0) return "$ 0,00";
  return "$ " + Number(n).toFixed(2).replace(".", ",").replace(/\B(?=(\d{3})+(?!\d))/g, ".");
}

// --- FUNCIÓN PRINCIPAL PARA GENERAR EL RECIBO ---
function generarReciboHistorico(nombreEmpleado, mes, anio) {
  try {
    var ss = SpreadsheetApp.openById(ID_RRHH);
    var hoja = ss.getSheetByName("CtaCte Empleados");
    var datos = hoja.getDataRange().getValues();
    var nombreBuscado = String(nombreEmpleado).toLowerCase().trim();
    var zona = ss.getSpreadsheetTimeZone();

    // 1. BUSCAR TELÉFONO EN COLUMNA I (Índice 8)
    var telefono = "";
    var hojaPersonal = ss.getSheetByName("Personal");
    
    if (hojaPersonal) {
      var datosP = hojaPersonal.getDataRange().getValues();
      for (var p = 1; p < datosP.length; p++) {
        // Columna A (0) es el nombre
        var nombreEnFila = String(datosP[p][0]).toLowerCase().trim();
        
        if (nombreEnFila == nombreBuscado) {
           // Columna I es el índice 8
           var rawTel = String(datosP[p][8] || ""); 
           
           // Dejar solo números
           telefono = rawTel.replace(/[^0-9]/g, "");
           
           // Si tiene 10 dígitos (ej: 2991234567), agregamos 549 (Argentina)
           if (telefono.length === 10) {
              telefono = "549" + telefono;
           }
           break; 
        }
      }
    }

    // 2. Definir fechas
    var mesNum = parseInt(mes);
    var anioNum = parseInt(anio);
    var mesActualStr = mesNum.toString().padStart(2, '0') + "-" + anioNum;
    
    var mesAntNum = mesNum - 1;
    var anioAntNum = anioNum;
    if (mesAntNum === 0) { mesAntNum = 12; anioAntNum--; }
    var mesAntStr = mesAntNum.toString().padStart(2, '0') + "-" + anioAntNum;

    // 3. Filtrar Historia
    var historia = [];
    for (var i = 1; i < datos.length; i++) {
      if (String(datos[i][1]).toLowerCase().trim() == nombreBuscado) {
        historia.push({
          fecha: datos[i][0],
          concepto: String(datos[i][2]),
          debe: Number(datos[i][3]) || 0,
          haber: Number(datos[i][4]) || 0
        });
      }
    }

    if (historia.length === 0) return { texto: "No hay historial.", telefono: "" };

    // 4. Buscar Cortes
    var idxCorteActual = -1;
    for (var j = historia.length - 1; j >= 0; j--) {
       var c = historia[j].concepto.toLowerCase();
       if ((c.includes("sueldo") || c.includes("pago") || c.includes("haberes")) && c.includes(mesActualStr)) {
          idxCorteActual = j;
          break;
       }
    }
    if (idxCorteActual === -1) idxCorteActual = historia.length - 1; 

    var idxCorteAnterior = -1;
    for (var j = idxCorteActual - 1; j >= 0; j--) {
       var c = historia[j].concepto.toLowerCase();
       if ((c.includes("sueldo") || c.includes("pago") || c.includes("haberes")) && c.includes(mesAntStr)) {
          idxCorteAnterior = j;
          break;
       }
    }

    // 5. Calcular Saldos
    var saldoAnterior = 0;
    for (var k = 0; k <= idxCorteAnterior; k++) {
       saldoAnterior += (historia[k].haber - historia[k].debe);
    }

    var itemsCiclo = [];
    var sumaIngresos = 0;
    var sumaEgresos = 0;

    for (var k = idxCorteAnterior + 1; k <= idxCorteActual; k++) {
       var m = historia[k];
       itemsCiclo.push(m);
       sumaIngresos += m.haber;
       sumaEgresos += m.debe;
    }

    var saldoFinalCiclo = saldoAnterior + sumaIngresos - sumaEgresos;

    // 6. Generar Texto
    var txt = "🧾 *HISTÓRICO: MES " + mesActualStr + "*\n";
    txt += "👤 " + nombreEmpleado.toUpperCase() + "\n";
    txt += "--------------------------------\n";

    txt += "*INGRESOS*\n";
    var hayIngresos = false;
    itemsCiclo.forEach(m => {
      if (m.haber > 0) {
        var fecha = Utilities.formatDate(new Date(m.fecha), zona, "dd/MM");
        txt += `${fecha} ${m.concepto.padEnd(18).substring(0,18)} ${formatearNumero(m.haber)}\n`;
        hayIngresos = true;
      }
    });
    if(!hayIngresos) txt += "(Sin ingresos)\n";
    txt += "--------------------------------\n";
    txt += `*TOTAL INGRESOS: ${formatearNumero(sumaIngresos)}*\n\n`;

    txt += "*DESCUENTOS / ADELANTOS*\n";
    var hayEgresos = false;
    itemsCiclo.forEach(m => {
      if (m.debe > 0) {
        var fecha = Utilities.formatDate(new Date(m.fecha), zona, "dd/MM");
        txt += `${fecha} ${m.concepto.padEnd(18).substring(0,18)} -${formatearNumero(m.debe).replace("$ ","")}\n`;
        hayEgresos = true;
      }
    });
    if(!hayEgresos) txt += "(Sin adelantos)\n";
    txt += "--------------------------------\n";
    txt += `*TOTAL EGRESOS: -${formatearNumero(sumaEgresos).replace("$ ","")}*\n\n`;

    txt += "================================\n";
    if (idxCorteAnterior > -1 || Math.abs(saldoAnterior) > 1) {
       txt += `Saldo Mes Ant (${mesAntStr}): ${formatearNumero(saldoAnterior)}\n`;
    }
    
    txt += `*NETO AL ${mesActualStr}:   ${formatearNumero(saldoFinalCiclo)}*\n`;
    txt += "================================";

    // Retorna Texto y Teléfono encontrado
    return { texto: txt, telefono: telefono };

  } catch (e) {
    return { texto: "Error: " + e.message, telefono: "" };
  }
}

      function toggleModoManual() {
        var isManual = document.getElementById("switchManual").checked;
        if(isManual) {
          document.getElementById("divSelector").style.display = "none";
          document.getElementById("divManual").style.display = "block";
          document.getElementById("selectEmpleado").value = ""; 
          document.getElementById("panelEmpleado").style.display = "none";
        } else {
          document.getElementById("divSelector").style.display = "block";
          document.getElementById("divManual").style.display = "none";
          document.getElementById("inputNombreManual").value = ""; 
          document.getElementById("panelEmpleado").style.display = "none";
        }

      function getNombreActual() {
        var isManual = document.getElementById("switchManual").checked;
        if(isManual) return document.getElementById("inputNombreManual").value.trim();
        else return document.getElementById("selectEmpleado").value;
      }

      function cargarLista() { 
        showLoading(true); 
        google.script.run.withSuccessHandler(function(lista) { 
           var s = document.getElementById("selectEmpleado"); 
           if (lista[0] && lista[0].startsWith("Error")) alert(lista[0]); 
           else lista.forEach(n => { var o = document.createElement("option"); o.text = n; o.value = n; s.add(o); }); 
           showLoading(false); 
        }).withFailureHandler(function(e){
           alert("Error de conexión: " + e.message); showLoading(false);
        }).obtenerListaEmpleados(); 
      }

      function cargarHistorial() {
        var nom = getNombreActual();
        if (!nom) return; 
        showLoading(true);
        document.getElementById("btnFiltroTodos").checked = true;
        google.script.run.withSuccessHandler(function(data) {
          if(data.error) { alert(data.error); showLoading(false); return; }
          var saldoEl = document.getElementById("txtTotal");
          saldoEl.innerText = formatMoney(data.total || 0);
          saldoEl.style.color = (data.total || 0) < 0 ? "#ffeaa7" : "white";
          datosGlobales = (data.movimientos || []); 
          datosGlobales.reverse(); 
          saldoAnteriorGlobal = data.saldoAnterior || 0; 
          renderizarLista("TODOS");
          document.getElementById("panelEmpleado").style.display = "block";
          showLoading(false);
        }).obtenerHistorialEmpleado(nom);
      }

      function aplicarFiltro(tipo) { renderizarLista(tipo); }

      function renderizarLista(filtro) {
        var ul = document.getElementById("listaMovimientos");
        ul.innerHTML = "";
        var listaMostrar = datosGlobales;
        if (filtro !== "TODOS") { listaMostrar = datosGlobales.filter(m => m.tipo === filtro); }
        if (listaMostrar.length === 0 && saldoAnteriorGlobal === 0) {
           ul.innerHTML = '<li class="list-group-item text-center text-muted py-4">No hay movimientos recientes.</li>'; return;
        }
        listaMostrar.forEach(m => {
          var claseMonto = m.tipo === "ACREDITADO" ? "text-acreditado" : "text-pagado";
          var signo = m.tipo === "ACREDITADO" ? "+ " : "- ";
          ul.innerHTML += `
            <li class="list-group-item d-flex justify-content-between align-items-center">
              <div><div class="fw-bold text-dark">${m.concepto}</div><small class="text-muted"><i class="bi bi-calendar3"></i> ${m.fecha} &nbsp;|&nbsp; ${m.medio}</small></div>
              <span class="${claseMonto} fs-5">${signo}${formatMoney(m.monto)}</span>
            </li>`;
        });
        if (filtro === "TODOS") {
            var colorSaldo = saldoAnteriorGlobal >= 0 ? "text-success" : "text-danger";
            var htmlSaldo = `
            <li class="list-group-item d-flex justify-content-between align-items-center bg-light" style="border-top: 2px dashed #ccc;">
              <div><div class="fw-bold text-secondary fst-italic"><i class="bi bi-clock-history"></i> Saldo Anterior al Ciclo</div><small class="text-muted">Acumulado histórico previo</small></div>
              <span class="${colorSaldo} fw-bold">${formatMoney(saldoAnteriorGlobal)}</span>
            </li>`;
            ul.innerHTML += htmlSaldo;
        }
      }

      function abrirImportar() { modalImport.show(); }
      function analizarPDF() {
         var txt = document.getElementById("txtPDF").value;
         if(txt.length < 50) { alert("Pega el texto del PDF primero."); return; }
         showLoading(true);
         google.script.run.withSuccessHandler(function(res){
            showLoading(false);
            if(res.status === "Error") { alert("Error en servidor. " + res.message); return; }
            if (res.data.empleadosEncontrados.length === 0) { alert("No se encontraron empleados válidos."); return; }
            document.getElementById("step1").style.display = "none";
            document.getElementById("step2").style.display = "block";
            document.getElementById("inputPeriodo").value = res.data.periodo || "11-2025";
            listaOpcionesGlobal = res.listaOpciones; datosPreview = res.data.empleadosEncontrados; 
            renderTablaImportacion();
         }).procesarTextoPDF(txt); 
      }
      function renderTablaImportacion() {
         var tbody = document.querySelector("#tablaPreview tbody");
         tbody.innerHTML = "";
         if (datosPreview.length === 0) { tbody.innerHTML = '<tr><td colspan="5" class="text-center py-4 text-danger">Sin datos.</td></tr>'; return; }
         datosPreview.forEach((item, idx) => {
            var tr = document.createElement("tr");
            var tdCheck = `<td><input type="checkbox" class="chk-item" data-idx="${idx}" ${item.nombreDB ? 'checked' : ''}></td>`;
            var tdNombrePDF = `<td><span class="fw-bold text-dark">${item.nombrePDF}</span></td>`;
            var tdCuil = `<td><span class="badge bg-light text-dark border font-monospace">${item.cuilPDF || 'N/A'}</span></td>`;
            var selectHTML = `<select class="form-select form-select-sm sel-empleado" data-idx="${idx}" style="font-weight:bold;"><option value="">❌ NO IMPORTAR</option>`;
            listaOpcionesGlobal.forEach(op => { var selected = (op === item.nombreDB) ? "selected" : ""; selectHTML += `<option value="${op}" ${selected}>${op}</option>`; });
            selectHTML += `</select>`;
            var estiloAlerta = !item.nombreDB ? "border-left: 3px solid red;" : "";
            var tdSys = `<td style="${estiloAlerta}">${selectHTML}</td>`;
            var tdMonto = `<td class="text-end fw-bold text-success fs-6">$ ${formatMoney(item.monto)}</td>`;
            tr.innerHTML = tdCheck + tdNombrePDF + tdCuil + tdSys + tdMonto;
            tbody.appendChild(tr);
         });
      }
      function confirmarImportacion() {
         var periodo = document.getElementById("inputPeriodo").value;
         var itemsFinales = [];
         var checks = document.querySelectorAll("#tablaPreview .chk-item");
         checks.forEach(chk => {
            if(chk.checked) {
               var idx = chk.getAttribute("data-idx");
               var select = document.querySelector(`.sel-empleado[data-idx="${idx}"]`);
               var nombreDB = select.value;
               var monto = datosPreview[idx].monto;
               if(nombreDB) itemsFinales.push({ nombreDB: nombreDB, monto: monto, importar: true });
            }
         });
         if(itemsFinales.length === 0) { alert("No hay empleados válidos seleccionados."); return; }
         if(!confirm(`¿Importar ${itemsFinales.length} sueldos para ${periodo}?`)) return;
         showLoading(true);
         var payload = JSON.stringify({ periodo: periodo, items: itemsFinales });
         google.script.run.withSuccessHandler(function(res){
            showLoading(false); alert(res); modalImport.hide(); cargarHistorial(); volverStep1();
         }).guardarImportacionSueldos(payload);
      }

      function showLoading(s) { document.getElementById("loading").style.display = s ? "flex" : "none"; }
