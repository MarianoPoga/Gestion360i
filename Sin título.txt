function onEditCompras(e) {
  // 1. Obtener la hoja y el rango que se acaba de editar
  var hojaActual = e.source.getActiveSheet();
  var rango = e.range;
  var fila = rango.getRow();
  var columna = rango.getColumn();

  // 2. VERIFICACIONES
  if (hojaActual.getName() !== "Compras" || columna !== 1 || fila < 9 || fila > 25) {
    return; 
  }

  // 3. Obtener el valor ingresado en A
  var valorBuscado = rango.getValue();

  // Si se borró el valor, limpiamos columnas B, D y F
  if (valorBuscado === "") {
    hojaActual.getRange(fila, 2).clearContent(); // Borra B
    hojaActual.getRange(fila, 4).clearContent(); // Borra D
    hojaActual.getRange(fila, 6).clearContent(); // Borra F
    return;
  }

  // 4. Buscar en la hoja "Proveedores"
  // (Le cambié el nombre a la variable para no confundirnos)
  var hojaProveedores = e.source.getSheetByName("Proveedores");
  if (!hojaProveedores) return; 

  var dataProv = hojaProveedores.getDataRange().getValues();
  var encontrado = false;

  // 5. Bucle Principal (Busca en Proveedores)
  for (var i = 0; i < dataProv.length; i++) {
    // dataProv[i][0] es Columna A de Proveedores
    if (dataProv[i][0] == valorBuscado) {
      
      var valorParaB = dataProv[i][1]; // Columna B (Nombre Proveedor)
      var valorParaC = dataProv[i][2]; // Columna C (Para inicial)
      var valorCrudoD = dataProv[i][3]; // Columna D (El dato intermedio para buscar diminutivo)
      
      // --- Lógica Inicial Mayúscula (Col F) ---
      var inicialC = "";
      if (valorParaC) {
        inicialC = valorParaC.toString().substring(0, 1).toUpperCase();
      }

      // --- Lógica Diminutivo (Col D) - SEGUNDA BÚSQUEDA ---
      var valorFinalD = ""; // Empezamos vacío
      
      if (valorCrudoD) {
        var hojaDatosRef = e.source.getSheetByName("Datos"); // Hoja donde están los diminutivos
        if (hojaDatosRef) {
          var dataRef = hojaDatosRef.getDataRange().getValues();
          
          // Bucle Secundario: Busca valorCrudoD en Col A de "Datos"
          for (var j = 0; j < dataRef.length; j++) {
            if (dataRef[j][0] == valorCrudoD) {
              valorFinalD = dataRef[j][1]; // Toma el valor de Col B (Diminutivo)
              break; // Ya lo encontramos, salimos del bucle pequeño
            }
          }
        }
      }
      // Si no encontró nada en "Datos", valorFinalD seguirá vacío o podés hacer que sea igual a valorCrudoD
      // if (valorFinalD === "") valorFinalD = valorCrudoD; // Descomenta esta linea si quieres que ponga el original si no encuentra diminutivo

      // 6. Escribir los resultados en "Compras"
      hojaActual.getRange(fila, 2).setValue(valorParaB);   // Columna B
      hojaActual.getRange(fila, 4).setValue(valorFinalD);  // Columna D (El diminutivo)
      hojaActual.getRange(fila, 6).setValue(inicialC);     // Columna F
      
      encontrado = true;
      break; 
    }
  }
}

function formatearStringComoDinero(valorString) {
  // 1. Intenta convertir la cadena a un número.
  var valorNumerico = parseFloat(valorString);

  // 2. Verifica si la conversión fue exitosa.
  if (isNaN(valorNumerico)) {
    return "Valor inválido"; // O manejar el error como prefieras.
  }

  // 3. Define el símbolo de moneda y el separador de miles.
  var simboloMoneda = ""; // Puedes cambiarlo a tu moneda local.
  var separadorMiles = ".";   // Común en Argentina. Usa "," si prefieres.
  var separadorDecimal = ","; // Común en Argentina. Usa "." si prefieres.
  var decimales = 2;         // Número de decimales.

  // 4. Formatea el número.
  var numeroFormateado = valorNumerico.toFixed(decimales).replace(/\d(?=(\d{3})+\D)/g, '$&' + separadorMiles).replace(".", separadorDecimal);

  // 5. Agrega el símbolo de moneda.
  return simboloMoneda + " " + numeroFormateado;
}

  if (isNaN(value)) value = 0;
  return value.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ".");

/**
 * Búsqueda de gastos en la hoja 'Compras' que deben ser imputados al cierre.
 */
function buscarGastosCompras(hojaCompras, etiquetaExacta, fechaFiltroISO, zona, startRow) {
  var listaGastos = [];
  var totalSalidas = 0;
  
  if (!hojaCompras) return { lista: listaGastos, total: totalSalidas };

  var lastRow = hojaCompras.getLastRow();
  var numRows = lastRow - startRow + 1;
  
  if (numRows <= 0) return { lista: listaGastos, total: totalSalidas };

  // Leemos columnas: A(0) Fecha, B(1) Prov, D(3) Detalle, E(4) Monto, F(5) Pago
  var datosC = hojaCompras.getRange(startRow, 1, numRows, 6).getValues(); 

  for (var i = 0; i < datosC.length; i++) {
    var pagoRaw = String(datosC[i][5]).trim();
    var pagoLimpio = pagoRaw.toLowerCase();
    var etiquetaLimpia = etiquetaExacta.toLowerCase();
    
    var esPendiente = (pagoLimpio === "caja" || pagoLimpio === "efectivo");
    var esAsignado = (pagoLimpio === etiquetaLimpia); // Comparación de la etiqueta limpia

    if (esPendiente || esAsignado) {
       var fRow = datosC[i][0];
       var monto = Number(datosC[i][4]) || 0;
       
       // Si es asignado O (es pendiente Y es de hoy)
       if (esAsignado) {
         totalSalidas += monto;
       } else if (esPendiente && fRow instanceof Date) {
         var fRowISO = Utilities.formatDate(fRow, zona, "yyyyMMdd");
         if (fRowISO === fechaFiltroISO) totalSalidas += monto;
       }
       
       // CORRECCIÓN VISUALIZACIÓN: Fecha completa dd/MM/yyyy
       var fechaItem = (fRow instanceof Date) ? Utilities.formatDate(fRow, zona, "dd/MM/yyyy") : "";
         
       listaGastos.push({
         id: startRow + i, // ID = Número de fila real en la hoja
         desc: String(datosC[i][1]) + " (" + String(datosC[i][3]) + ")", // Proveedor (Detalle)
         monto: monto,
         fecha: fechaItem,
         obligatorio: esAsignado
       });
    }
  }

  return { lista: listaGastos, total: totalSalidas };
}

/**
 * Búsqueda de adelantos en la hoja 'CtaCte Empleados' asignados al cierre.
 */
function buscarAdelantos(hojaCtaCte, etiquetaExacta, zona, startRow) {
  var listaAdelantosEfectivo = [];
  var listaAdelantosMerc = [];
  var totalAdelantosEfectivo = 0;

  if (!hojaCtaCte) return { listaEfectivo: listaAdelantosEfectivo, listaMerc: listaAdelantosMerc, total: totalAdelantosEfectivo };

  var lastRow = hojaCtaCte.getLastRow();
  var numRows = lastRow - startRow + 1;
  
  if (numRows <= 0) return { listaEfectivo: listaAdelantosEfectivo, listaMerc: listaAdelantosMerc, total: totalAdelantosEfectivo };

  // Leemos: A(0) Fecha, B(1) Empleado, C(2) Concepto, D(3) Debe (Monto), F(5) Medio (Etiqueta)
  var datosA = hojaCtaCte.getRange(startRow, 1, numRows, 6).getValues(); 
  var etiquetaLimpia = etiquetaExacta.toLowerCase();

  for (var j = 0; j < datosA.length; j++) {
    var medioRaw = String(datosA[j][5]).trim().toLowerCase();
    
    if (medioRaw === etiquetaLimpia) {
       var fRow = datosA[j][0];
       var concepto = String(datosA[j][2]);
       var debe = Number(datosA[j][3]) || 0; 
       
       // CORRECCIÓN VISUALIZACIÓN: Fecha completa dd/MM/yyyy
       var fechaItem = (fRow instanceof Date) ? Utilities.formatDate(fRow, zona, "dd/MM/yyyy") : "";
       
       var itemAdelanto = {
         id: startRow + j,
         desc: String(datosA[j][1]) + " (" + concepto + ")", // Empleado (Concepto)
         monto: debe,
         fecha: fechaItem,
         obligatorio: true
       };

       if (concepto.toLowerCase().includes("mercader")) {
         listaAdelantosMerc.push(itemAdelanto);
       } else {
         listaAdelantosEfectivo.push(itemAdelanto);
         totalAdelantosEfectivo += debe;
       }
    }
  }

  return { listaEfectivo: listaAdelantosEfectivo, listaMerc: listaAdelantosMerc, total: totalAdelantosEfectivo };
}


/**

 */
function obtenerItemsPorTurno(turnoSeleccionado, fechaCierreStr) {
  var resultado = { 
    items: [], prefill: {}, totalSalidasCaja: 0, 
    listaGastosCaja: [], listaAdelantosEfectivo: [], listaAdelantosMerc: [], error: null
  }; 

  try {
    var ssMov = SpreadsheetApp.openById(ID_MOVIMIENTOS); 
    var ssRRHH = SpreadsheetApp.openById(ID_RRHH);       
    var zona = ssMov.getSpreadsheetTimeZone(); // ¡Uso consistente de la zona!

    // --- 1. LEER CONFIG (Items de Venta) ---
    var hojaItems = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("ItemVentas");
    if (!hojaItems) throw new Error("Falta hoja ItemVentas");
    var datosItems = hojaItems.getDataRange().getValues();
    var encabezados = datosItems[0]; 
    var turnoBuscado = String(turnoSeleccionado).toLowerCase().trim();
    var colIndex = encabezados.map(function(t) { return String(t).toLowerCase().trim(); }).indexOf(turnoBuscado);
    
    if (colIndex === -1) throw new Error("Turno no encontrado en ItemVentas.");
    for (var r = 1; r < datosItems.length; r++) {
      var concepto = datosItems[r][0]; var activado = datosItems[r][colIndex];
      var esVerdadero = (activado === true || String(activado).toUpperCase() === "TRUE" || String(activado).toUpperCase() === "VERDADERO");
      if (concepto && esVerdadero) resultado.items.push(concepto);
    }

    // --- 2. PREPARAR FECHAS Y ETIQUETAS ---
    var fechaObj;
    if (fechaCierreStr) { 
      var p = fechaCierreStr.split("-"); 
      // Creamos la fecha en la zona horaria del script, luego la forzamos a ser medianoche local
      fechaObj = new Date(p[0], p[1]-1, p[2]); 
    } else { 
      fechaObj = new Date(); 
    }
    
    var fechaFiltroISO = Utilities.formatDate(fechaObj, zona, "yyyyMMdd");
    var fechaVisual = Utilities.formatDate(fechaObj, zona, "dd/MM/yyyy");
    
    // Etiqueta exacta: "caja dd/mm/aaaa turno"
    var etiquetaExacta = "caja " + fechaVisual + " " + turnoBuscado;
    
    // Rango de búsqueda optimizado
    var startRow = 2; // Asumimos que leemos los últimos 300 movimientos
    
    // --- 3. BUSCAR GASTOS (Compras) ---
    var hojaCompras = ssMov.getSheetByName("Compras");
    if (hojaCompras && hojaCompras.getLastRow() >= startRow) {
        var startRowCompras = Math.max(startRow, hojaCompras.getLastRow() - 300 + 1);
        var resGastos = buscarGastosCompras(hojaCompras, etiquetaExacta, fechaFiltroISO, zona, startRowCompras);
        resultado.listaGastosCaja = resGastos.lista;
        resultado.totalSalidasCaja += resGastos.total;
    }

    // --- 4. BUSCAR ADELANTOS (RRHH) ---
    var hojaCtaCte = ssRRHH.getSheetByName("CtaCte Empleados");
    if (hojaCtaCte && hojaCtaCte.getLastRow() >= startRow) {
        var startRowAdelantos = Math.max(startRow, hojaCtaCte.getLastRow() - 300 + 1);
        var resAdelantos = buscarAdelantos(hojaCtaCte, etiquetaExacta, zona, startRowAdelantos);
        resultado.listaAdelantosEfectivo = resAdelantos.listaEfectivo;
        resultado.listaAdelantosMerc = resAdelantos.listaMerc;
        resultado.totalSalidasCaja += resAdelantos.total; // Sumar solo adelantos en EFECTIVO
    }


  } catch (e) { 
    resultado.error = "Error al buscar movimientos: " + e.message;
    console.error(resultado.error);
  }

  return resultado;
}

// --- GUARDAR CIERRE (CORREGIDA LA LECTURA DE VALORES DEL FORMULARIO) ---
function guardarCierre(formObject) {
  var zona = getZone(); // ¡Uso consistente de la zona!
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Ventas");
    if (!hoja) throw new Error("Error: Hoja Ventas no encontrada.");
    
    var ultimaColumna = hoja.getLastColumn();
    if (ultimaColumna < 1) return "Error: Hoja Ventas sin encabezados.";

    var encabezadosRaw = hoja.getRange(1, 1, 1, ultimaColumna).getValues()[0];
    var encabezadosLimpios = encabezadosRaw.map(function(t) { return String(t).toLowerCase().trim(); });
    
    var nuevaFila = new Array(ultimaColumna).fill(null); 
    var idxFecha = encabezadosLimpios.indexOf("fecha");
    var idxTurno = encabezadosLimpios.indexOf("turno");
    var idxTotal = encabezadosLimpios.indexOf("total");
    
    // Preparar Fecha
    var fechaGuardar = new Date(); 
    if (formObject.fechaCierre) {
      var partes = formObject.fechaCierre.split("-");
      var ahora = new Date(); // Obtenemos la hora actual
      
      // Creamos la fecha usando el día seleccionado PERO inyectamos la hora actual
      // partes[1] - 1 porque los meses en JS van de 0 a 11
      fechaGuardar = new Date(partes[0], partes[1] - 1, partes[2], ahora.getHours(), ahora.getMinutes(), ahora.getSeconds());
    }

    if (idxFecha > -1) nuevaFila[idxFecha] = fechaGuardar;
    if (idxTurno > -1) nuevaFila[idxTurno] = formObject.turnoHidden;
    
    var montoEfectivoCierre = 0;
    var desgloseItems = []; 
    var sumaTotal = 0;

    // A. Mapear valores del formulario a la nueva fila (ITERAMOS SOBRE LOS ENCABEZADOS DE LA HOJA)
    for (var i = 0; i < encabezadosRaw.length; i++) {
        var keyHoja = encabezadosRaw[i]; // Nombre original del encabezado (Ej: "Efectivo")
        var keyLimpia = encabezadosLimpios[i]; // Nombre limpio de la hoja (Ej: "efectivo")
        
        // Excluir claves de control o no relevantes para la suma/registro directo
        if (keyLimpia === "fecha" || keyLimpia === "turno" || keyLimpia === "total") {
            continue; 
        }
        
        // Intentamos encontrar la clave en el formObject. Los nombres de campo del formulario 
        // DEBEN COINCIDIR con los encabezados de la hoja.
        var val = parseFloat(formObject[keyHoja] || formObject[keyLimpia]); // Intentamos con nombre original y limpio
        
        if (!isNaN(val)) {
            nuevaFila[i] = val;
            
            if (keyLimpia.includes("efectivo")) montoEfectivoCierre = val;
            
            // Desglose y suma
            if (val !== 0) {
                desgloseItems.push({ item: keyHoja, monto: val });
            }
            sumaTotal += val;
        } else {
             // Si no es un número válido, lo dejamos como null (o 0 si prefieres que aparezca)
             nuevaFila[i] = 0; 
        }
    }
    
    // B. Asignar Total Calculado
    if (idxTotal > -1) {
      nuevaFila[idxTotal] = sumaTotal;
    } 
    
    // --- 1. GUARDAR EN HOJA VENTAS ---
    hoja.appendRow(nuevaFila);
    var lastRow = hoja.getLastRow();
    

    // --- 2. GUARDAR EN RENDICIONES (CÓDIGO RESTAURADO) ---------------------
    var hojaRendiciones = ss.getSheetByName("Rendiciones");
    var montoEfectivo = parseFloat(formObject.Efectivo) || 0;
    if (sumaTotal > 0) {
        var desc = "Cierre " + formObject.turnoHidden;
        
        // Asiento de DÉBITO (Entrada de dinero / Caja)
        var filaDebe = [
            fechaGuardar, 
            desc, 
            montoEfectivo, // DEBE
            "",        // HABER
            "Ventas"     // Categoría
        ];
        hojaRendiciones.appendRow(filaDebe);
        

        // --- Aplicar formatos de moneda y fecha a las filas recién insertadas en Rendiciones
        var lastRowRendiciones = hojaRendiciones.getLastRow();
        var rangeRendiciones = hojaRendiciones.getRange(lastRowRendiciones - 1, 1, 2, 4);
        
        // Columna 1 (Fecha): CORRECCIÓN AQUÍ PARA GUARDAR FECHA Y HORA
        // Usamos formato completo para que se vea también la hora si existe en el objeto Date
        rangeRendiciones.offset(0, 0, 2, 1).setNumberFormat("dd/mm/yyyy hh:mm");
        
        // Columnas 3 y 4 (Debe y Haber)
        rangeRendiciones.offset(0, 2, 2, 2).setNumberFormat("$ #,##0.00");
    }
    // ---



    // --- 3. IMPUTACIÓN DE COMPRAS (Gastos) ---
    if (formObject.gastosIds) {
      var ids = formObject.gastosIds.split(",");
      var hojaCompras = ss.getSheetByName("Compras");
      
      var turnoNombre = String(formObject.turnoHidden).toLowerCase();
      var letraTurno = turnoNombre.includes("tarde") ? "T" : (turnoNombre.includes("delivery") ? "D" : (turnoNombre.includes("noche") ? "N" : "M"));
      
      var fechaStr = Utilities.formatDate(fechaGuardar, zona, "dd/MM/yyyy");
      var nuevoEstado = "Caja " + fechaStr + " " + letraTurno;

      ids.forEach(function(fila) {
        var filaNum = parseInt(fila);
        if (!isNaN(filaNum) && filaNum > 1 && hojaCompras) {
          hojaCompras.getRange(filaNum, 6).setValue(nuevoEstado); 
        }
      });
    }

    // --- 4. NOTIFICACIÓN PUSHOVER ---
    var totalFormato = formatCurrencyForPushover(sumaTotal);
    var title =  "$" + totalFormato + " - " + Utilities.formatDate(fechaGuardar, zona, "dd/MM/yyyy") + " - " + formObject.turnoHidden;
    
    var body = "CIERRE REGISTRADO:\n";
    desgloseItems.forEach(function(item) {
        body += item.item + ": $" + formatCurrencyForPushover(item.monto) + "\n";
    });
    body += "\nTOTAL FINAL: $" + totalFormato;

    enviarNotificacionPushover(title, body);
    
    // [Se mantienen los formatos]
    // CORRECCIÓN AQUÍ TAMBIÉN: Formato en hoja Ventas con hora
    if (idxFecha > -1) hoja.getRange(lastRow, idxFecha + 1).setNumberFormat("dd/mm/yyyy hh:mm");
    if (ultimaColumna > 2) hoja.getRange(lastRow, 3, 1, ultimaColumna - 2).setNumberFormat("$ #,##0.00");

    return "✅ Cierre guardado. Total: $" + sumaTotal.toLocaleString('es-AR', { minimumFractionDigits: 2 });

  } catch (e) { 
    console.error("Error en guardarCierre: " + e.message);
    return "❌ Error al guardar cierre: " + e.message; 
  }
}

function obtenerUltimosCierres() {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Ventas");
    var zona = ss.getSpreadsheetTimeZone(); // ¡Uso consistente de la zona!
    
    var lastRow = hoja.getLastRow();
    if (lastRow < 2) return []; 
    
    var filasALeer = 30; 
    var inicio = Math.max(2, lastRow - filasALeer + 1);
    var datos = hoja.getRange(inicio, 1, lastRow - inicio + 1, hoja.getLastColumn()).getValues();
    var encabezados = hoja.getRange(1, 1, 1, hoja.getLastColumn()).getValues()[0];
    var encabezadosLimp = encabezados.map(function(t) { return String(t).toLowerCase().trim(); });

    var idxFecha = encabezadosLimp.indexOf("fecha");
    var idxTurno = encabezadosLimp.indexOf("turno");
    
    if (idxFecha === -1 || idxTurno === -1) return [];

    var lista = [];
    
    for (var i = datos.length - 1; i >= 0; i--) {
      var fechaRaw = datos[i][idxFecha];
      var turno = datos[i][idxTurno];
      var fechaFmt = "";
      var fechaISO = ""; 

      if (fechaRaw instanceof Date) {
        // CORRECCIÓN VISUALIZACIÓN: Usar dd/MM/yyyy en lugar de día de semana
        fechaFmt = Utilities.formatDate(fechaRaw, zona, "dd/MM/yyyy");
        fechaISO = Utilities.formatDate(fechaRaw, zona, "yyyy-MM-dd");
      } else {
        fechaFmt = String(fechaRaw);
      }
      
      if (fechaFmt && turno) {
        lista.push({ fecha: fechaFmt, fechaISO: fechaISO, turno: turno });
      }
    }
    return lista;
  } catch (e) { return [{ fecha: "Error", turno: e.message }]; }
}

/**
 * Función auxiliar para verificar si la caja está cerrada.
 */
function esTurnoCerrado(fechaStr, turno) { // Recibe fechaStr (ISO: YYYY-MM-DD)
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Ventas");
    if (!hoja || hoja.getLastRow() < 2) return false;

    var zona = ss.getSpreadsheetTimeZone();
    
    var partes = fechaStr.split("-");
    var fechaObj = new Date(partes[0], partes[1] - 1, partes[2]);
    
    var fechaStrCheck = Utilities.formatDate(fechaObj, zona, "yyyyMMdd");
    var turnoCheck = String(turno).toLowerCase().trim();

    // Leemos las últimas 50 filas
    var lastRow = hoja.getLastRow();
    var startRow = Math.max(2, lastRow - 50);
    var datos = hoja.getRange(startRow, 1, lastRow - startRow + 1, 2).getValues(); 
    var encabezados = hoja.getRange(1, 1, 1, 2).getValues()[0].map(function(t) { return String(t).toLowerCase().trim(); });
    var idxFecha = encabezados.indexOf("fecha");
    var idxTurno = encabezados.indexOf("turno");

    if (idxFecha === -1 || idxTurno === -1) return false;

    for (var i = 0; i < datos.length; i++) {
      var fRow = datos[i][idxFecha];
      var tRow = String(datos[i][idxTurno]).toLowerCase().trim();
      
      if (fRow instanceof Date) {
         var fRowStr = Utilities.formatDate(fRow, zona, "yyyyMMdd");
         // Si coincide fecha y turno, es que YA se cerró
         if (fRowStr === fechaStrCheck && tRow === turnoCheck) {
           return true; 
         }
      }
    }
    return false;
  } catch (e) { 
    console.error("Error en esTurnoCerrado: " + e.message);
    return false; 
  }
}

      // --- Contador de Billetes Lógica ---
      function generarFilasContador() {
        var container = document.getElementById("billetesContainer");
        container.innerHTML = "";
        
        DENOMINACIONES.forEach(den => {
          var row = document.createElement("div"); row.className = "billete-row";
          row.innerHTML = `
            <div class="billete-denominacion">$ ${new Intl.NumberFormat('es-AR').format(den)}</div>
            <input type="number" min="0" value="0" class="form-control billete-input billete-cantidad me-3" 
                   id="count_${den}" oninput="calcularTotalContador()" data-denominacion="${den}">
            <div class="billete-total" id="subtotal_${den}">$ 0</div>
          `;
          container.appendChild(row);
        });
        
        calcularTotalContador(); 
      }
      
      function calcularTotalContador() {
        var total = 0;
        DENOMINACIONES.forEach(den => {
          var count = parseInt(document.getElementById(`count_${den}`).value) || 0;
          var subtotal = count * den;
          total += subtotal;
          document.getElementById(`subtotal_${den}`).innerText = "$ " + new Intl.NumberFormat('es-AR').format(subtotal);
        });
        document.getElementById("txtTotalContador").innerText = "$ " + new Intl.NumberFormat('es-AR', { minimumFractionDigits: 2 }).format(total);
        return total;
      }
      
      function abrirModalContador() {
        // Al abrir, si el input de efectivo tiene un valor, se podría intentar pre-cargar el conteo.
        // Aquí solo se resetea por simplicidad:
        document.getElementById("billetesContainer").querySelectorAll('input').forEach(input => input.value = 0);
        calcularTotalContador(); 
        
        modalContador.show();
      }
      
      function transferirTotal() {
        var total = calcularTotalContador();
        var inputEfectivo = document.getElementById("inputEfectivo");
        if (inputEfectivo) inputEfectivo.value = total > 0 ? total.toFixed(2) : ""; // Aseguramos 2 decimales
        modalContador.hide();

      function llenarSelectEmp(lista) {
        var sel = document.getElementById("selectEmpModal");
        if(lista && lista.length && !lista[0].startsWith("Error")) {
           lista.forEach(n => { var op = document.createElement("option"); op.text=n; op.value=n; sel.add(op); });
        }

      function actualizarHistorial() {
        document.getElementById("loadingHistorial").style.display = "inline-block"; 
        
        google.script.run.withSuccessHandler(function(lista) {
          historialGlobal = lista || []; 
          
          document.getElementById("loadingHistorial").style.display = "none";
          var divLista = document.getElementById("listaHistorial");
          divLista.innerHTML = "";
          
          if (!lista || lista.length === 0) {
            divLista.innerHTML = '<div class="text-center text-muted p-2 small">No hay datos recientes.</div>';
            return;
          }

          lista.slice(0, 10).forEach(function(item) {
            var turnoLower = String(item.turno || "").toLowerCase();
            var color = "secondary";
            
            if (turnoLower.includes("mañana")) color = "success";
            else if (turnoLower.includes("tarde")) color = "danger";
            else if (turnoLower.includes("delivery")) color = "warning text-dark";
            else if (turnoLower.includes("noche")) color = "primary";

            divLista.innerHTML += `
              <div class="historial-item d-flex justify-content-between align-items-center">
                <span class="fw-bold text-dark">${item.fecha}</span>
                <span class="badge bg-${color} rounded-pill shadow-sm" style="width:80px">${item.turno}</span>
              </div>
            `;
          });
          
          validarCierreExistente();
          
        }).obtenerUltimosCierres();

      function validarCierreExistente() {

        alerta.style.display = "none";

        if (existe) {
          alerta.style.display = "block";
          cont.style.display = "none";
          btnDiv.style.display = "none";

      function crearElementoLista(obj, funcName, tipo) {
         var row = document.createElement("div"); row.className = "gasto-check-item";
         var deleteAction = (tipo === "gasto") ? `borrarCompra(${obj.id})` : `borrarItemAdelanto(${obj.id})`;
         var btnDelete = '';
         
         if (tipo === "adelantoMerc") {
            btnDelete = `<button type="button" class="btn btn-sm btn-outline-danger border-0 ms-2" onclick="${deleteAction}" title="Eliminar"><i class="bi bi-trash"></i></button>`;
         }

         if (obj.obligatorio) {
             row.style.backgroundColor = "#f8f9fa"; row.style.cursor = "default";
             row.innerHTML = `
               <input class="form-check-input" type="checkbox" value="${obj.monto}" checked disabled style="opacity:0.6">
               <div class="gasto-desc text-start ms-2">
                  ${obj.fecha ? `<span class="badge bg-light text-secondary border me-1" style="font-size: 0.75em;">${obj.fecha}</span>` : ''}
                  ${obj.desc}
               </div>
               <div class="gasto-monto text-end ms-2 me-3 text-nowrap">$ ${new Intl.NumberFormat('es-AR').format(obj.monto)}</div>
               ${btnDelete}
             `;
         } else {
             row.className += " clickable";
             row.onclick = function(e) { if (e.target.type !== 'checkbox') { var chk = this.querySelector('input[type="checkbox"]'); chk.checked = !chk.checked; chk.dispatchEvent(new Event('change')); }};
             var dataAttr = (tipo === "gasto") ? `data-gasto-id="${obj.id}"` : `data-adelanto-id="${obj.id}"`;
             
             row.innerHTML = `
               <input class="form-check-input ${tipo === 'gasto' ? 'gasto-check' : 'adelanto-check'}" type="checkbox" value="${obj.monto}" ${dataAttr} onchange="${funcName}()" checked>
               <div class="gasto-desc text-start ms-2">
                  ${obj.fecha ? `<span class="badge bg-light text-secondary border me-1" style="font-size: 0.75em;">${obj.fecha}</span>` : ''}
                  ${obj.desc}
               </div>
               <div class="gasto-monto text-end ms-2 text-nowrap">$ ${new Intl.NumberFormat('es-AR').format(obj.monto)}</div>
             `;
         }
         return row;
      }

      function recalcularCompras() {
        var container = document.getElementById('listaContainer_COMPRAS');
        if(!container) return;
        var checks = container.querySelectorAll('input[type="checkbox"]'); 
        var total = 0; var ids = [];
        checks.forEach(chk => { 
           if (chk.checked) {
              total += parseFloat(chk.value || 0); 
              if(!chk.disabled) ids.push(chk.getAttribute("data-gasto-id"));
           }
        });
        var inp = document.getElementById("inputComprasTotal");
        if (inp) inp.value = total > 0 ? total.toFixed(2) : "";
        document.getElementById("gastosIds").value = ids.join(",");
      }

      function recalcularAdelantos() {
        var contCaja = document.getElementById('listaContainer_CAJA');
        if(contCaja) {
           var totalCaja = 0;
           contCaja.querySelectorAll('input[type="checkbox"]').forEach(chk => { if(chk.checked) totalCaja += parseFloat(chk.value || 0); });
           var inpCaja = document.getElementById("inputAdelantosCaja");
           if(inpCaja) inpCaja.value = totalCaja > 0 ? totalCaja.toFixed(2) : "";
        }
        var contMerc = document.getElementById('listaContainer_MERC');
        if(contMerc) {
           var totalMerc = 0;
           contMerc.querySelectorAll('input[type="checkbox"]').forEach(chk => { if(chk.checked) totalMerc += parseFloat(chk.value || 0); });
           var inpMerc = document.getElementById("inputAdelantosMerc");
           if(inpMerc) inpMerc.value = totalMerc > 0 ? totalMerc.toFixed(2) : "";
        }
      }

  function abrirModalCalculadora(idInputDestino) {
    inputDestinoId = idInputDestino;
    limpiarCalculadora();
    
   // --- NUEVO: Traer valor actual del input ---
    var input = document.getElementById(inputDestinoId);
    if (input && input.value) {
        var valorActual = parseFloat(input.value);
        // Si hay un numero valido y mayor a 0, lo agregamos de una vez
        if (!isNaN(valorActual) && valorActual > 0) {
            agregarItemVisual(valorActual);
        }
    }
    modalCalculadora.show();
    // Ponemos foco en el input del modal para escribir rápido
    setTimeout(() => document.getElementById("calcInputMonto").focus(), 500);
  }

  function agregarACalculadora() {
    var input = document.getElementById("calcInputMonto");
    var valor = parseFloat(input.value);
    
    if (isNaN(valor) || valor <= 0) return;

    agregarItemVisual(valor);
    
    input.value = "";
    input.focus();
  }

  function agregarItemVisual(monto) {
    totalCalculadora += monto;
    actualizarTotalVisual();

    var ul = document.getElementById("calcListaItems");
    var li = document.createElement("li");
    li.className = "list-group-item d-flex justify-content-between align-items-center py-1 px-3";
    li.innerHTML = `
        <span class="text-dark fw-bold">+ $ ${new Intl.NumberFormat('es-AR').format(monto)}</span>
        <button class="btn btn-link text-danger p-0 text-decoration-none" onclick="borrarItemCalculadora(this, ${monto})"><i class="bi bi-x-lg"></i></button>
    `;
    ul.prepend(li); // Agregar al principio de la lista
  }

  function borrarItemCalculadora(btn, monto) {
    totalCalculadora -= monto;
    if(totalCalculadora < 0) totalCalculadora = 0;
    actualizarTotalVisual();
    btn.closest("li").remove();
  }

  function actualizarTotalVisual() {
    document.getElementById("calcTxtTotal").innerText = "$ " + new Intl.NumberFormat('es-AR', {minimumFractionDigits: 2}).format(totalCalculadora);
  }

  function limpiarCalculadora() {
    totalCalculadora = 0;
    document.getElementById("calcListaItems").innerHTML = "";
    document.getElementById("calcInputMonto").value = "";
    document.getElementById("calcFooterMsg").style.display = "none";
    actualizarTotalVisual();
  }

  function confirmarCalculo() {

  }

/**
 * Registra el Adelanto (DEBE) en la CtaCte del Empleado y, opcionalmente, el Ingreso (HABER)
 * si el usuario marcó la opción "Registrar Ingreso".
 */
function guardarAdelantoDoble(formObject) {
  var ssRRHH = SpreadsheetApp.openById(ID_RRHH); 
  var ssMov = SpreadsheetApp.openById(ID_MOVIMIENTOS); 
  var zona = ssRRHH.getSpreadsheetTimeZone(); 

  try {
    // 1. Obtener y validar datos
    var fecha = new Date(); // Captura Fecha Y HORA actual
    var empleado = String(formObject.empleadoFinal).trim();
    var monto = parseFloat(formObject.monto) || 0;
    var detalleUsuario = String(formObject.detalle).trim();
    var origen = formObject.origen; 
    var registrarIngreso = formObject.registrarIngreso; 

    if (!empleado) return "Error: Falta seleccionar empleado.";
    if (monto <= 0) return "Error: El monto debe ser mayor a cero.";

    // 2. Preparar hojas
    var hojaCtaCte = ssRRHH.getSheetByName("CtaCte Empleados"); 
    if (!hojaCtaCte) {
      hojaCtaCte = ssRRHH.insertSheet("CtaCte Empleados");
      hojaCtaCte.appendRow(["Fecha", "Empleado", "Concepto", "Debe (Adelantos)", "Haber (Sueldos)", "Medio/Origen"]);
    }
    
    // 3. Construir Medio de Pago y Validar Cierre (Si origen es Caja)
    var medioPago = origen; 
    var turnoAsignado = formObject.turnoAsignado;

    if (origen === "Caja") {
       if (!turnoAsignado) return "Error: Si el origen es Caja, debes elegir un turno.";
       
       // Validación de cierre
       var fechaStrISO = Utilities.formatDate(fecha, zona, "yyyy-MM-dd"); 
       if (typeof esTurnoCerrado === 'function' && esTurnoCerrado(fechaStrISO, turnoAsignado)) {
         return "Error: La Caja del turno '" + turnoAsignado + "' ya fue cerrada. No puedes agregar movimientos.";
       }

       var fechaStr = Utilities.formatDate(fecha, zona, "dd/MM/yyyy");
       medioPago = "Caja " + fechaStr + " " + turnoAsignado;
    }

    // --- REGISTRO PRINCIPAL ---
    
    // A. REGISTRO DEL ADELANTO (DEBE)
    var conceptoAdelanto = "Adelanto - " + detalleUsuario;
    hojaCtaCte.appendRow([fecha, empleado, conceptoAdelanto, monto, 0, medioPago]);

    // B. REGISTRO DEL INGRESO (HABER)
    if (registrarIngreso) {
      var conceptoIngreso = "Sueldo " + detalleUsuario;
      hojaCtaCte.appendRow([fecha, empleado, conceptoIngreso, 0, monto, "-"]);
    }
    
    // Aplicar formatos a las filas añadidas (FECHA Y HORA)
    var filasAgregadas = registrarIngreso ? 2 : 1;
    var lr = hojaCtaCte.getLastRow();
    // CAMBIO: Formato con hora "dd/mm/yyyy hh:mm"
    hojaCtaCte.getRange(lr - filasAgregadas + 1, 1, filasAgregadas, 1).setNumberFormat("dd/mm/yyyy hh:mm");
    hojaCtaCte.getRange(lr - filasAgregadas + 1, 4, filasAgregadas, 2).setNumberFormat("$ #,##0.00");


    // 4. IMPACTO EN MOVIMIENTOS (Solo si es Rendición)
    if (origen === "Rendición") {
      var hojaRend = ssMov.getSheetByName("Rendiciones");
      if (!hojaRend) hojaRend = ssMov.insertSheet("Rendiciones");
      
      var detalleCaja = "Adelanto: " + empleado + " (" + detalleUsuario + ")";
      // Escribe en HABER (Egreso/Salida)
      hojaRend.appendRow([fecha, detalleCaja, 0, monto, "RRHH"]);
      
      // CAMBIO: Formato en Rendiciones también
      var lrRend = hojaRend.getLastRow();
      hojaRend.getRange(lrRend, 1).setNumberFormat("dd/mm/yyyy hh:mm");
      hojaRend.getRange(lrRend, 3, 1, 2).setNumberFormat("$ #,##0.00");
      
      var msg = `✅ Adelanto registrado (Descontado de Rendiciones).`;
      if (registrarIngreso) msg += ` Se registró el ingreso de Sueldo/Bono.`;
      return msg;
      
    } else {
      var msg = `✅ Adelanto registrado (Se descontará al cerrar la caja ${turnoAsignado}).`;
      if (registrarIngreso) msg += ` Se registró el ingreso de Sueldo/Bono.`;
      return msg;
    }

  } catch (e) { 
    Logger.log("Error en guardarAdelantoDoble: " + e.message);
    return "Error al registrar: " + e.message; 
  }
}


function guardarAdelanto(form) {
  try {

    var empleado = form.empleadoFinal;

    var detalleUsuario = form.detalle;

    if (!empleado || empleado.trim() === "") return "Error: Falta seleccionar empleado.";

      if (hojaRend) {
        // CORRECCIÓN: USAMOS "Gasto (Rendiciones)" PARA UNIFICAR CRITERIO
        var detalleRend = "Gasto (Rendiciones): " + proveedor + " - " + detalle;
        
        // Haber (Salida)
        hojaRend.appendRow([fecha, detalleRend, 0, monto, "Compras"]);
        hojaRend.getRange(hojaRend.getLastRow(), 1).setNumberFormat("dd/MM/yyyy");
        hojaRend.getRange(hojaRend.getLastRow(), 3, 1, 2).setNumberFormat("$ #,##0.00");
      }
    }

    actualizarListaProveedores(proveedor, tipo, detalle, pago);

    return { 
   status: "OK", 
   mensaje: "✅ Gasto registrado.", 
   listasActualizadas: obtenerHistorialCompleto() // <--- CAMBIO AQUÍ
    };

  } catch (e) { return { error: e.message }; }
}

// --- 4. ELIMINAR COMPRA ---
function eliminarCompra(idFila) {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hojaCompras = ss.getSheetByName("Compras");
    var fila = parseInt(idFila);
    
    if (!hojaCompras || fila < 2) return "Error: Fila no válida.";
    if (fila > hojaCompras.getLastRow()) return "Error: La fila ya no existe.";

    // Leer datos antes de borrar
    var datosFila = hojaCompras.getRange(fila, 1, 1, 6).getValues()[0];
    var fechaCompra = datosFila[0];
    var proveedorCompra = String(datosFila[1]);
    var detalleCompra = String(datosFila[3]); 
    var montoCompra = parseFloat(datosFila[4]);
    var pagoCompra = String(datosFila[5]).toLowerCase();

    // 1. BORRAR DE CTA CTE PROVEEDORES
    if (pagoCompra.includes("cta") || pagoCompra.includes("corriente")) {
       eliminarDeCtaCteProveedor(ss, fechaCompra, proveedorCompra, montoCompra);
    }

    // 2. BORRAR DE RENDICIONES
    else if (pagoCompra.includes("rendic")) {
       eliminarDeRendiciones(ss, fechaCompra, montoCompra, proveedorCompra, detalleCompra);
    }

    // 3. BORRAR DE COMPRAS
    try {
      hojaCompras.deleteRow(fila);
      return "OK";
    } catch(err) { return "Error Excel: " + err.message; }

  } catch (e) { return "Error crítico: " + e.message; }
}

// --- AUXILIAR: BORRAR DE RENDICIONES ---
function eliminarDeRendiciones(ss, fechaTarget, montoTarget, provTarget, detalleTarget) {
  var hoja = ss.getSheetByName("Rendiciones");
  if (!hoja) return;
  
  var datos = hoja.getDataRange().getValues();
  var zona = ss.getSpreadsheetTimeZone();
  var fechaTargetStr = Utilities.formatDate(new Date(fechaTarget), zona, "yyyyMMdd");
  
  // CORRECCIÓN: USAMOS EL MISMO STRING QUE AL GUARDAR
  var detalleBuscado = "Gasto (Rendiciones): " + provTarget + " - " + detalleTarget;

  // Estructura Rendiciones: A=Fecha, B=Detalle, D=Haber
  for (var i = 1; i < datos.length; i++) {
    var fechaRow = datos[i][0];
    var detalleRow = String(datos[i][1]);
    var haberRow = parseFloat(datos[i][3]); 
    
    if (fechaRow instanceof Date) {
      var fechaRowStr = Utilities.formatDate(fechaRow, zona, "yyyyMMdd");
      
      // Coincidencia exacta
      if (fechaRowStr === fechaTargetStr && haberRow === montoTarget && detalleRow === detalleBuscado) {
        hoja.deleteRow(i + 1); 
        break; 
      }
    }
  }
}

// --- AUXILIAR: BORRAR DE CTA CTE ---
function eliminarDeCtaCteProveedor(ss, fechaTarget, provTarget, montoTarget) {
  var hoja = ss.getSheetByName("CtaCte Proveedores");
  if (!hoja) return;
  var datos = hoja.getDataRange().getValues();
  var zona = ss.getSpreadsheetTimeZone();
  var fechaTargetStr = Utilities.formatDate(new Date(fechaTarget), zona, "yyyyMMdd");
  var provTargetLimpio = provTarget.toLowerCase().trim();

  for (var i = 1; i < datos.length; i++) {
    var fechaRow = datos[i][0];
    var provRow = String(datos[i][1]).toLowerCase().trim();
    var haberRow = parseFloat(datos[i][5]); 
    
    if (fechaRow instanceof Date) {
      var fechaRowStr = Utilities.formatDate(fechaRow, zona, "yyyyMMdd");
      if (fechaRowStr === fechaTargetStr && provRow === provTargetLimpio && haberRow === montoTarget) {
        hoja.deleteRow(i + 1);
        break;
      }
    }
  }
}

// --- 1. OBTENER DATOS (PROVEEDORES Y LISTAS) ---
function obtenerDatosCompras() {
  try {
    var ssConfig = SpreadsheetApp.getActiveSpreadsheet(); 
    
    // ... (Tu lógica de carga de proveedores y listas) ...
    
    var hojaProv = ssConfig.getSheetByName("Proveedores");
    var listaProveedores = {};
    if (hojaProv && hojaProv.getLastRow() > 1) {
      var datosProv = hojaProv.getRange(2, 1, hojaProv.getLastRow()-1, 4).getValues();
      for (var i = 0; i < datosProv.length; i++) {
        var nombre = String(datosProv[i][0]).trim();
        if (nombre) {
          listaProveedores[nombre] = { tipo: datosProv[i][1], detalle: datosProv[i][2], pago: datosProv[i][3] };
        }
      }
    }

    var hojaDatos = ssConfig.getSheetByName("Datos");
    var opciones = { detalles: [], pagos: [] };
    if (hojaDatos) {
      var encabezados = hojaDatos.getRange(1, 1, 1, hojaDatos.getLastColumn()).getValues()[0];
      var colDetalle = encabezados.indexOf("Detalle");
      var colPago = encabezados.indexOf("Forma de Pago");
      var lastRow = hojaDatos.getLastRow();
      if (lastRow > 1) {
        var datosD = hojaDatos.getRange(2, 1, lastRow-1, hojaDatos.getLastColumn()).getValues();
        for (var j = 0; j < datosD.length; j++) {
          if (colDetalle > -1 && datosD[j][colDetalle]) opciones.detalles.push(datosD[j][colDetalle]);
          if (colPago > -1 && datosD[j][colPago]) opciones.pagos.push(datosD[j][colPago]);

      }

    return { proveedores: listaProveedores, listas: opciones };

  } catch (e) { return { error: e.message }; }
}

// --- 2. OBTENER HISTORIAL (HOY + 30 DÍAS CON ORDENAMIENTO) ---
function obtenerHistorialCompleto(sortType) {
  try {
    var ssRes = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ssRes.getSheetByName("Compras");
    if (!hoja) return { hoy: [], antiguas: [] };

    var lastRow = hoja.getLastRow();
    if (lastRow < 2) return { hoy: [], antiguas: [] };

    // Leemos las últimas 500 filas
    var startRow = Math.max(2, lastRow - 500);
    var numRows = lastRow - startRow + 1;
    var datos = hoja.getRange(startRow, 1, numRows, 6).getValues(); 
    
    var zona = Session.getScriptTimeZone();
    var hoyObj = new Date();
    var hoyStr = Utilities.formatDate(hoyObj, zona, "yyyyMMdd");
    
    // Fecha límite (30 días atrás)
    var limiteObj = new Date();
    limiteObj.setDate(limiteObj.getDate() - 30);
    var limiteStr = Utilities.formatDate(limiteObj, zona, "yyyyMMdd");

    var listaHoy = [];
    var listaAntiguas = [];

    // Recorremos de abajo hacia arriba (más nuevo primero)
    for (var i = datos.length - 1; i >= 0; i--) {
      var fechaRow = datos[i][0];
      
      if (fechaRow instanceof Date) {
        var fechaStr = Utilities.formatDate(fechaRow, zona, "yyyyMMdd");
        var fechaVisual = Utilities.formatDate(fechaRow, zona, "dd/MM");
        
        var item = {
          id: startRow + i, // ID Real
          proveedor: datos[i][1],
          tipo: datos[i][2], 
          detalle: datos[i][3],
          monto: datos[i][4],
          pago: datos[i][5],
          fecha: fechaVisual,
          fechaRaw: fechaRow // Guardamos el objeto Date para ordenar correctamente
        };

        if (fechaStr === hoyStr) {
          listaHoy.push(item);
        } else if (fechaStr >= limiteStr && fechaStr < hoyStr) {
          listaAntiguas.push(item);
        }
      }
    }
    
    // 3. LÓGICA DE ORDENAMIENTO PARA ANTIGUAS
    if (sortType === 'proveedor') {
        listaAntiguas.sort(function(a, b) {
            return String(a.proveedor).localeCompare(String(b.proveedor));
        });
    } else {
        // Ordenar por fecha (default) - Se hace aquí porque el loop es LIFO. 
        // Para asegurar que sea por fecha descendente, ordenamos por la propiedad fechaRaw.
        listaAntiguas.sort(function(a, b) {
            return b.fechaRaw.getTime() - a.fechaRaw.getTime();
        });
    }
    
    // Limpiamos las propiedades crudas antes de enviar
    listaAntiguas.forEach(item => delete item.fechaRaw);
    listaHoy.forEach(item => delete item.fechaRaw);
    
    return { hoy: listaHoy, antiguas: listaAntiguas };

  } catch (e) { return { hoy: [], antiguas: [], error: e.message }; }
}

function obtenerFacturasPendientes() {
  try {
    var ssRes = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ssRes.getSheetByName("Compras");
    if (!hoja || hoja.getLastRow() < 2) return [];

    var datos = hoja.getDataRange().getValues();
    var listaPendientes = [];
    var zona = Session.getScriptTimeZone();

    // Buscamos las columnas por nombre para mayor seguridad
    var encabezados = datos[0];
    var colFecha = 0; // Asumimos Col A
    var colProveedor = 1; // Asumimos Col B
    var colDetalle = 3; // Asumimos Col D
    var colMonto = 4; // Asumimos Col E
    var colFactura = 6; // Col G: Estado Factura

    for (var i = 1; i < datos.length; i++) {
      var estadoFactura = String(datos[i][colFactura]).trim();
      
      if (estadoFactura === "Pendiente") {
        var fechaRaw = datos[i][colFecha];
        
        listaPendientes.push({
          fecha: (fechaRaw instanceof Date) ? Utilities.formatDate(fechaRaw, zona, "dd/MM/yyyy") : "",
          proveedor: datos[i][colProveedor],
          detalle: datos[i][colDetalle],
          monto: datos[i][colMonto],
          id: i + 1 // Fila real
        });
      }
    }
    
    // Devolvemos lo más nuevo primero
    return listaPendientes.reverse(); 

  } catch (e) { return { error: "Error obteniendo pendientes: " + e.message }; }
}

function obtenerVistaRendiciones() {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hojaCaja = ss.getSheetByName("Rendiciones");
    
    var saldoCiclo = 0;
    var listaMovimientos = [];

    if (hojaCaja && hojaCaja.getLastRow() > 1) {
      var datos = hojaCaja.getDataRange().getValues();
      
      // Recorremos hacia atrás hasta encontrar "Retiro Total"
      for (var i = datos.length - 1; i >= 1; i--) {
        var detalle = String(datos[i][1]);
        
        if (detalle.includes("Retiro Total")) break; // CORTAMOS AQUÍ

        var fechaRow = datos[i][0];
        var debe = Number(datos[i][2]) || 0;
        var haber = Number(datos[i][3]) || 0;
        var cat = datos[i][4] || "Gral";

        saldoCiclo += (debe - haber);

        if (fechaRow instanceof Date) {
           var tipoMov = (debe > 0) ? "INGRESO" : "EGRESO";
           var montoVisual = (debe > 0) ? debe : haber;
           var fechaFmt = Utilities.formatDate(fechaRow, Session.getScriptTimeZone(), "dd/MM/yyyy");

           listaMovimientos.push({
             fecha: fechaFmt,
             detalle: detalle,
             monto: montoVisual,
             tipo: tipoMov,
             categoria: cat
           });
        }
      }
    }

    return {
      saldo: saldoCiclo,
      movimientos: listaMovimientos // Ya vienen ordenados
    };

  } catch (e) { return { saldo: 0, movimientos: [], error: e.message }; }
}

function obtenerTareas() {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Tareas");
    if (!hoja) return [];
    
    var lastRow = hoja.getLastRow();
    if (lastRow < 2) return [];

    // Leemos todo para filtrar
    var datos = hoja.getRange(2, 1, lastRow - 1, 5).getValues();
    var lista = [];
    var zona = Session.getScriptTimeZone();

    // Recorremos de abajo (más nuevo) hacia arriba
    for (var i = datos.length - 1; i >= 0; i--) {
      // ID = Fila real (i + 2)
      var filaId = i + 2;
      var estado = String(datos[i][4]).toLowerCase();
      
      // Solo mostramos las PENDIENTES o las REALIZADAS de hoy (para que no desaparezcan al instante)
      // O simplemente traemos las ultimas 10 sin importar, pero priorizamos pendientes
      
      lista.push({
        id: filaId,
        fecha: (datos[i][0] instanceof Date) ? Utilities.formatDate(datos[i][0], zona, "dd/MM") : "",
        caracter: datos[i][1],
        tarea: datos[i][2],
        usuario: datos[i][3],
        estado: datos[i][4] // "Pendiente" o "Realizada"
      });
      
      if (lista.length >= 15) break; // Limite visual
    }
    
    // Ordenar: Pendientes primero
    lista.sort(function(a, b) {
      return (a.estado === "Pendiente" ? -1 : 1);
    });

    return lista;
  } catch (e) { return []; }
}

function guardarNuevaTarea(form) {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Tareas");
    if (!hoja) { hoja = ss.insertSheet("Tareas"); /* headers... */ }
    
    var fecha = new Date();
    var caracter = form.caracterTarea;
    var texto = form.textoTarea;
    var usuario = "Empleado"; // O puedes poner un input para nombre
    
    hoja.appendRow([fecha, caracter, texto, usuario, "Pendiente"]);
    
    return "OK";
  } catch (e) { return "Error: " + e.message; }
}

function cambiarEstadoTarea(idFila) {
  try {
    var ss = SpreadsheetApp.openById(ID_MOVIMIENTOS);
    var hoja = ss.getSheetByName("Tareas");
    var fila = parseInt(idFila);
    
    var estadoActual = hoja.getRange(fila, 5).getValue();
    var nuevoEstado = (estadoActual === "Pendiente") ? "Realizada" : "Pendiente";
    
    hoja.getRange(fila, 5).setValue(nuevoEstado);
    
    // Si se marca realizada, se pone fecha de finalización? Opcional.
    
    return "OK";
  } catch (e) { return "Error: " + e.message; }
}

function doPost(e) {
  // 1. Preflight / CORS (Opcional si usas 'Content-Type: text/plain' en el cliente)
  // Pero generalmente doPost maneja la lógica.
  
  var output = { status: "ERROR", message: "Solicitud inválida" };
  
  try {
    // 2. Parsear el cuerpo de la solicitud JSON
    if (!e.postData || !e.postData.contents) {
      throw new Error("No se recibieron datos JSON.");
    }
    
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var params = data.params || {};
    
    // 3. Router de Acciones
    switch (action) {
      // --- TAREAS ---
      case "obtenerTareas":
        output = { status: "OK", data: obtenerTareas() };
        break;
      case "guardarNuevaTarea":
        // params: { caracterTarea, textoTarea }
        output = { status: "OK", data: guardarNuevaTarea(params) };
        break;
      case "cambiarEstadoTarea":
        // params: { idFila }
        output = { status: "OK", data: cambiarEstadoTarea(params.idFila) };
        break;

      // --- COMPRAS / GASTOS ---
      case "obtenerDatosCompras":
        output = { status: "OK", data: obtenerDatosCompras() };
        break;
      case "obtenerHistorialCompleto":
        // params: { sortType }
        output = { status: "OK", data: obtenerHistorialCompleto(params.sortType) };
        break;
      case "guardarCompra":
        // params: { formulario... }
        output = { status: "OK", data: guardarCompra(params) }; 
        break;
      case "eliminarCompra":
        // params: { id }
        output = { status: "OK", data: eliminarCompra(params.id) };
        break;
      case "obtenerFacturasPendientes":
        output = { status: "OK", data: obtenerFacturasPendientes() };
        break;
      case "marcarFacturaEntregada":
        // params: { id }
        output = { status: "OK", data: marcarFacturaEntregada(params.id) };
        break;

      // --- RENDICIONES ---
      case "obtenerVistaRendiciones":
        output = { status: "OK", data: obtenerVistaRendiciones() };
        break;

      // --- ADELANTOS ---
      case "obtenerEmpleadosParaSelector":
        output = { status: "OK", data: obtenerEmpleadosParaSelector() };
        break;
      case "guardarAdelantoDoble":
        // params: { formulario... }
        // Nota: Asegúrate de enviar los campos esperados por guardarAdelantoDoble formObject
        output = { status: "OK", data: guardarAdelantoDoble(params) };
        break;

      // --- CIERRE ---
      case "obtenerUltimosCierres":
        output = { status: "OK", data: obtenerUltimosCierres() };
        break;
      case "obtenerItemsPorTurno":
        // params: { fecha, turno }
        output = { status: "OK", data: obtenerItemsPorTurno(params.fecha, params.turno) };
        break;
      case "guardarCierre":
        // params: { formulario... }
        output = { status: "OK", data: guardarCierre(params) };
        break;
        
      // --- EXTRAS ---
      case "test":
        output = { status: "OK", message: "Conexión exitosa con GAS API" };
        break;

      default:
        output = { status: "ERROR", message: "Acción desconocida: " + action };
    }

  } catch (error) {
    output = { status: "ERROR", message: error.toString() };
  }

  // 4. Devolver JSON
  return ContentService.createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}
